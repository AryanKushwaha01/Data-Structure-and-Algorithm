void rec(vector<int>& nums, vector<vector<int>>& ans,vector<int>& temp, int i){
        // vector<int> temp;
        if(i == nums.size()){
            ans.push_back({temp});
            return;
        }

        //Include the element
        temp.push_back(nums[i]);
        rec(nums, ans, temp, i+1);

	temp.pop_back();

        //Exclude the element
        rec(nums, ans, temp, i+1);
    }
    vector<vector<int>> subsets(vector<int>& nums) {
        vector<vector<int>> ans;
        vector<int> temp;
        rec(nums, ans, temp, 0);
        return ans;
    }
