class Solution {
public:
    int ans = 0;
    int cnt = 0;
    void Inorder(TreeNode* root, int k){
        if(root == NULL) return;

        //Left call
        Inorder(root->left, k);

        cnt++;
        if(cnt == k){
            ans = root -> val;
            return;
        }
	//Right call
        Inorder(root -> right, k);
    }
    int kthSmallest(TreeNode* root, int k) {
        Inorder(root, k);
        return ans;
    }
};
